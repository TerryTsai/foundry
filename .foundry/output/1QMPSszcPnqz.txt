const BOUNDED_THRESHOLD = 10_240 // 10 KB
const boundedOutput = z.object({
  content:    z.string(),
  outputFile: z.string().optional(),
  totalLines: z.number().optional(),
  totalBytes: z.number().optional(),
  error:      z.string().optional(),
})
// ── Helpers ──
function esc(s: string): string {
  return "'" + s.replace(/'/g, "'\\''") + "'"
}
// Resolve a tool-supplied path inside the container. Absolute paths are
// used as-is so the agent can read /tmp, /proc, /etc, or any other readable
// path in the sandbox. Relative paths are rooted at /work (the workspace
// bind-mount) for ergonomics: most tool calls target the project source.
// An empty/undefined path defaults to /work.
function workPath(p?: string): string {
  if (!p) return '/work'
  if (p.startsWith('/')) return p
  return `/work/${p}`
}
function bounded(workspace: string, raw: string): { content: string; outputFile?: string; totalLines?: number; totalBytes?: number } {
  const bytes = Buffer.byteLength(raw, 'utf-8')
  if (bytes <= BOUNDED_THRESHOLD) return { content: raw }
  const outputDir = join(workspace, '.foundry', 'output')
  mkdirSync(outputDir, { recursive: true })
  const filename = `${nanoid(12)}.txt`
  writeFileSync(join(outputDir, filename), raw, 'utf-8')
  const lines = raw.split('\n')
  return {
    content:    lines.slice(0, 20).join('\n'),
    outputFile: `.foundry/output/${filename}`,
    totalLines: lines.length,
    totalBytes: bytes,
  }
}
/** Write a temp file into the workspace (visible to container via bind mount). */
function writeTmp(workspace: string, content: string, ext = 'txt'): string {
  const relPath = `.foundry/.tmp/${nanoid(8)}.${ext}`
  const absPath = join(workspace, relPath)
  mkdirSync(dirname(absPath), { recursive: true })
  writeFileSync(absPath, content, 'utf-8')
  return relPath
}
function runCommand(
  db: Db, config: Config, sessionId: string, workspace: string,
  command: string, toolName: string, toolInput: unknown, network = false,
): Promise<{ stdout: string; stderr: string; exitCode: number }> {
  const { emitter } = domain.executionCreate(db, config, {
    command, session_id: sessionId, tool_name: toolName, tool_input: toolInput, network, workspace,
  })
  const stdout: string[] = []
  const stderr: string[] = []
  return new Promise(resolve => {
    emitter.on('stdout', (line: string) => stdout.push(line))
    emitter.on('stderr', (line: string) => stderr.push(line))
    emitter.once('exit', (code: number) => {
      resolve({ stdout: stdout.join('\n'), stderr: stderr.join('\n'), exitCode: code })
    })
  })
}
async function runApi(
  db: Db, config: Config, sessionId: string, workspace: string, command: string, toolName: string, toolInput: unknown,
): Promise<{ content: string; outputFile?: string; totalLines?: number; totalBytes?: number; error?: string }> {
  try {
    const r = await runCommand(db, config, sessionId, workspace, command, toolName, toolInput, true)
    if (r.exitCode === 0) return bounded(workspace, r.stdout)
    return { content: '', error: `request failed (exit ${r.exitCode})` }
  } catch (err) {
    return { content: '', error: err instanceof Error ? err.message : String(err) }
  }
}

export function makeTools(
  db: Db, config: Config, sessionId: string, projectId: string, ticketId: string, agentName: string, workspace: string, enabledTools?: string[],
) {
  const api = `http://host.containers.internal:${config.port}`
  const all = {
    // ── exec ──
    exec: tool({
      description:  'Run a shell command in a sandboxed container with the project repo at /work. Large stdout is written to a file; use the read tool to paginate.',
      inputSchema:  validate.toolExec,
      outputSchema: boundedOutput.extend({ stderr: z.string(), exitCode: z.number() }),
      execute: async (args) => {
        const command = args.cwd ? `cd ${esc(args.cwd)} && ${args.command}` : args.command
        try {
          const r = await runCommand(db, config, sessionId, workspace, command, 'exec', args, args.network ?? false)
          return { ...bounded(workspace, r.stdout), stderr: r.stderr, exitCode: r.exitCode }
        } catch (err) {
          return { content: '', stderr: '', exitCode: -1, error: err instanceof Error ? err.message : String(err) }
        }
      },
    }),
    // ── read (pager — does NOT go through bounded) ──
    read: tool({
      description:  'Read the contents of a file from the workspace. Use offset and limit to paginate. Large output is written to a file you can read with offset/limit.',
      inputSchema:  validate.toolRead,
      outputSchema: boundedOutput,
      execute: async (args: { path: string; offset?: number; limit?: number }) => {
        const path = esc(workPath(args.path))
        const command = (args.offset !== undefined || args.limit !== undefined)
          ? `sed -n '${(args.offset ?? 0) + 1},${(args.offset ?? 0) + (args.limit ?? 1_000_000)}p' ${path}`
          : `cat ${path}`
        try {
          const r = await runCommand(db, config, sessionId, workspace, command, 'read', args)
          if (r.exitCode !== 0) return { content: '', error: r.stderr || 'file not found or unreadable' }
          return bounded(workspace, r.stdout)
        } catch (err) {
          return { content: '', error: err instanceof Error ? err.message : String(err) }
        }
      },
    }),
    // ── write ──
    write: tool({
      description:  'Create or overwrite a file in the workspace.',
      inputSchema:  validate.toolWrite,
      outputSchema: z.object({ ok: z.boolean(), error: z.string().optional() }),
      execute: async (args) => {
        const tmpRel = writeTmp(workspace, args.content)
        const dest = esc(workPath(args.path))
        const command = `mkdir -p "$(dirname ${dest})" && mv /work/${esc(tmpRel)} ${dest}`
        try {
          const r = await runCommand(db, config, sessionId, workspace, command, 'write', { path: args.path })
          return r.exitCode === 0 ? { ok: true } : { ok: false, error: r.stderr || 'write failed' }
        } catch (err) {
          return { ok: false, error: err instanceof Error ? err.message : String(err) }
        }
      },
    }),
    // ── patch ──
    patch: tool({
      description:  'Apply a targeted edit to a file by replacing an exact string match.',
      inputSchema:  validate.toolPatch,
      outputSchema: z.object({ ok: z.boolean(), error: z.string().optional() }),
      execute: async (args) => {
        const meta = writeTmp(workspace, JSON.stringify({
          path: workPath(args.path),
          old:  args.old_string,
          new:  args.new_string,
        }), 'json')
        const command = `python3 -c '
import json, os, sys
m = json.load(open("/work/${meta}"))
os.unlink("/work/${meta}")
c = open(m["path"]).read()
if m["old"] not in c:
    print("old_string not found in file", file=sys.stderr); sys.exit(1)
open(m["path"],"w").write(c.replace(m["old"],m["new"],1))
print("ok")
'`
        try {
          const r = await runCommand(db, config, 
sessionId, workspace, 
command, 'patch', { path: args.path })
          return r.exitCode === 0 ? { ok: true } : { ok: false, error: r.stderr || r.stdout || 'patch failed' }
        } catch (err) {
          return { ok: false, error: err instanceof Error ? err.message : String(err) }
        }
      },
    }),
    // ── search ──
    search: tool({
      description:  'Search file contents using grep with regex support. Large output is written to a file you can read with offset/limit.',
      inputSchema:  validate.toolSearch,
      outputSchema: boundedOutput,
      execute: async (args) => {
        const searchPath = workPath(args.path)
        const parts = ['grep', '-rn', '--color=never']
        if (args.glob) parts.push('--include', esc(args.glob))
        parts.push('-e', esc(args.pattern), esc(searchPath))
        // || true so exit 1 (no matches) doesn't mark execution as failed
        const command = `${parts.join(' ')} | sed 's|/work/||g' || true`
        try {
          const r = await runCommand(db, config, sessionId, workspace, command, 'search', args)
          const output = r.stdout.trimEnd()
          if (!output) return { content: '' }
          return bounded(workspace, output)
        } catch (err) {
          return { content: '', error: err instanceof Error ? err.message : String(err) }
        }
      },
    }),
    // ── find ──
    find: tool({
      description:  'List files matching a glob pattern in the workspace. Defaults to max_depth 3 for performance — increase or omit to scan deeper. If a deep scan times out, narrow the path or pattern. Large output is written to a file you can read with offset/limit.',
      inputSchema:  validate.toolFind,
      outputSchema: boundedOutput,
      execute: async (args) => {
        const depth = args.max_depth ?? 3
        const searchPath = workPath(args.path)
        const parts = ['find', esc(searchPath), '-maxdepth', String(depth), '-type', 'f', '-name', esc(args.pattern), '-print']
        const command = `${parts.join(' ')} | sed 's|/work/||g'`
        try {
          const r = await runCommand(db, config, sessionId, workspace, command, 'find', args)
          const output = r.stdout.trim()
          if (!output) return { content: '' }
          return bounded(workspace, output)
        } catch (err) {
          return { content: '', error: err instanceof Error ? err.message : String(err) }
        }
      },
    }),
    // ── fetch ──
    fetch: tool({
      description:  'Make an HTTP request and return the response. Large responses are written to a file; use the read tool to paginate.',
      inputSchema:  validate.toolFetch,
      outputSchema: boundedOutput.extend({ status: z.number() }),
      execute: async (args) => {
        const curlArgs = ['-s', '-w', '\\n%{http_code}', '-X', args.method ?? 'GET']
        if (args.headers) {
          for (const [k, val] of Object.entries(args.headers)) curlArgs.push('-H', `${k}: ${val}`)
        }
        if (args.body) curlArgs.push('-d', args.body)
        curlArgs.push(args.url)
        const command = `curl ${curlArgs.map(a => esc(a)).join(' ')}`
        try {
          const r = await runCommand(db, config, sessionId, workspace, command, 'fetch', args, true)
          const lines = r.stdout.trimEnd().split('\n')
          const statusCode = toInt(lines.pop() ?? '0')
          return { ...bounded(workspace, lines.join('\n')), status: statusCode }
        } catch (err) {
          return { content: '', status: 0, error: err instanceof Error ? err.message : String(err) }
        }
      },
    }),
    // ── scrape ──
    scrape: tool({
      description:  'Extract clean readable text from a web page. Large output is written to a file you can read with offset/limit.',
      inputSchema:  validate.toolScrape,
      outputSchema: boundedOutput,
      execute: async (args) => {
        const url = esc(args.url)
        const command = `lynx -dump -nolist -width=120 ${url} 2>/dev/null || curl -sL ${url} | sed 's/<[^>]*>//g' | sed '/^$/d'`
        try {
          const r = await runCommand(db, config, sessionId, workspace, command, 'scrape', args, true)
          return bounded(workspace, r.stdout.trim())
        } catch (err) {
          return { content: '', error: err instanceof Error ? err.message : String(err) }
        }
      },
    }),
    // ── think ──
    think: tool({
      description:  'Use this tool to reason, plan, or analyze before taking action. No side effects.',
      inputSchema:  validate.toolThink,
      outputSchema: z.object({ ok: z.boolean() }),
      execute: async () => ({ ok: true }),
    }),
    // ── agents ──
    agents: tool({
      description:  'List all agents in the current project. Useful for targeting sub-tickets at a specific agent.',
      inputSchema:  z.object({}),
      outputSchema: boundedOutput,
      execute: async () => {
        const command = `curl -sf '${api}/agent?projectId=${projectId}'`
        return runApi(db, config, sessionId, workspace, command, 'agents', {})
      },
    }),
    // ── ticket ──
    ticket: tool({
      description:  'Create, read, or list tickets in the current project. Use create to spawn sub-tasks, optionally assigning to an agent by ID. Use the agents tool first to discover agent IDs.',
      inputSchema:  validate.toolTicket,
      outputSchema: boundedOutput,
      execute: async (args) => {
        let command: string
        if (args.action === 'read') {
          if (!args.id) return { content: '', error: 'id is required for read' }
          command = `curl -sf '${api}/ticket/${args.id}'`
        } else if (args.action === 'list') {
          command = `curl -sf '${api}/ticket?projectId=${projectId}'`
        } else if (args.action === 'create') {
          if (!args.title || !args.body) return { content: '', error: 'title and body are required for create' }
          const payload = JSON.stringify({
            project_id: projectId, title: args.title, body: args.body,
            priority: 0, parent_id: ticketId, created_by: agentName,
            ...(args.agent_id ? { agent_id: args.agent_id } : {}),
          })
          command = `curl -sf -X POST -H 'Content-Type: application/json' -d ${esc(payload)} '${api}/ticket'`
        } else {
          return { content: '', error: 'unknown action' }
        }
        return runApi(db, config, sessionId, workspace, command, 'ticket', args)
      },
    }),
  }
  if (!enabledTools || enabledTools.length === 0) return all
  const allRec = all as Record<string, (typeof all)[keyof typeof all]>
  const filtered: typeof allRec = {}
  for (const name of enabledTools) {
    if (name in allRec) filtered[name] = allRec[name]
  }
  return filtered
}

// MCP server for the Claude Agent SDK. The model sees these as
// `mcp__foundry__<name>` and the SDK loop drives them, but each tool's
// execution still goes through podman via runCommand → domain.executionCreate →
// spawnRun. Same isolation, same execution-row tracking, same bounded output.
function asText(value: unknown): { content: [{ type: 'text'; text: string }] } {
  const text = typeof value === 'string' ? value : JSON.stringify(value)
  return { content: [{ type: 'text', text }] }
}
